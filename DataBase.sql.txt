create database PaymentManager;
CREATE USER 'PaymentManager'@'localhost' IDENTIFIED BY 'PMAD';
GRANT ALL PRIVILEGES ON PaymentManager.* TO 'PaymentManager'@'localhost';
FLUSH PRIVILEGES;




-- Loại chi tiêu
CREATE TABLE kind (
    id INT AUTO_INCREMENT PRIMARY KEY,  -- Mã loại chi (khóa chính)
    beneficiary VARCHAR(255) NOT NULL   -- Người thụ hưởng
);

drop table payment;
-- Tạo bảng chi tiêu
CREATE TABLE payment (
    id INT AUTO_INCREMENT PRIMARY KEY,           -- Mã chi tiêu (khóa chính)
    kindId INT NULL REFERENCES kind(id),         -- Loại chi (có thể NULL)
    amount INT NOT NULL,                         -- Số tiền chi tiêu
    paymentDay DATETIME DEFAULT NOW(),           -- Thời gian chi tiêu
    note TEXT                                    -- Ghi chú  
);
